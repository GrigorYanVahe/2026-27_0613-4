Node.js Скрипт Сканирования Папок

Если браузер блокирует чтение глубоких вложенных файлов, запустите этот скрипт прямо в папке со студентами. Он сгенерирует файл results.json, который можно загрузить в панель. 


const fs = require('fs');
const path = require('path');

const ROOT_DIR = './'; // Путь к вашей главной папке
const SUBJECTS = {
  'dynamic_web': 4,
  'graphic_Interfaces': 2,
  'java_script': 5
};

const students = [];
const items = fs.readdirSync(ROOT_DIR, { withFileTypes: true });

for (const item of items) {
  if (item.isDirectory()) {
    const studentName = item.name;
    const studentData = { name: studentName, subjects: {} };

    for (const [subject, taskCount] of Object.entries(SUBJECTS)) {
      studentData.subjects[subject] = [];
      for (let i = 1; i <= taskCount; i++) {
        const taskFolder = path.join(ROOT_DIR, studentName, subject, `result-${i}`);
        let isDone = false;
        let grade = null;
        const isGrade = f => /^grade(\.txt)+$/i.test(f);
        try {
          if (fs.existsSync(taskFolder)) {
            const files = fs.readdirSync(taskFolder);
            isDone = files.some(f => !isGrade(f)); // Сдано, если есть что-то кроме grade.txt
            const gradeFile = files.find(isGrade);
            if (gradeFile) {
              const m = fs.readFileSync(path.join(taskFolder, gradeFile), 'utf8').match(/\d+(?:[.,]\d+)?/);
              const n = m ? parseFloat(m[0].replace(',', '.')) : NaN;
              grade = (n >= 0 && n <= 10) ? n : 'invalid';
            }
          }
        } catch (e) {}
        studentData.subjects[subject].push({ task: `result-${i}`, done: isDone, grade });
      }
    }
    students.push(studentData);
  }
}

fs.writeFileSync('results.json', JSON.stringify(students, null, 2));
console.log('Готово! Результаты сохранены в results.json');
